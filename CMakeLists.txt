cmake_minimum_required(VERSION 3.16)
project(ghostline_net_tycoon VERSION 0.3.3 LANGUAGES CXX)
find_package(SDL2 CONFIG QUIET)
if(TARGET SDL2::SDL2)
  set(GHOSTLINE_SDL_TARGET SDL2::SDL2)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  set(GHOSTLINE_SDL_TARGET PkgConfig::SDL2)
endif()
add_executable(ghostline src/ghostline.cpp)
target_compile_features(ghostline PRIVATE cxx_std_17)
target_compile_definitions(ghostline PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(ghostline PRIVATE ${GHOSTLINE_SDL_TARGET})
if(WIN32)
  add_executable(Ghostline-Installer WIN32 installer/windows_installer.cpp)
  target_compile_features(Ghostline-Installer PRIVATE cxx_std_17)
  target_link_libraries(Ghostline-Installer PRIVATE urlmon shell32 ole32)
endif()
